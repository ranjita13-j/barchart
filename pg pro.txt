// Data files
$studentFile = "students.json";
$paymentFile = "payments.json";

// Create files if they don't exist
if (!file_exists($studentFile)) {
    file_put_contents($studentFile, json_encode([]));
}

if (!file_exists($paymentFile)) {
    file_put_contents($paymentFile, json_encode([]));
}

// Read data
$students = json_decode(file_get_contents($studentFile), true);
$payments = json_decode(file_get_contents($paymentFile), true);

$message = "";


// ===============================
// ADD STUDENT
// ===============================

if (isset($_POST["add_student"])) {

    $student = [
        "id" => time(),
        "name" => $_POST["name"],
        "phone" => $_POST["phone"],
        "room" => $_POST["room"],
        "rent" => (float)$_POST["rent"],
        "due_day" => (int)$_POST["due_day"]
    ];

    $students[] = $student;

    file_put_contents(
        $studentFile,
        json_encode($students, JSON_PRETTY_PRINT)
    );

    $message = "Student added successfully!";
}


// ===============================
// PAY RENT
// ===============================

if (isset($_POST["pay_rent"])) {

    $studentId = $_POST["student_id"];
    $amount = (float)$_POST["amount"];

    foreach ($students as $student) {

        if ($student["id"] == $studentId) {

            $payment = [
                "id" => time(),
                "student_id" => $student["id"],
                "name" => $student["name"],
                "amount" => $amount,
                "payment_date" => date("Y-m-d"),
                "rent_month" => date("Y-m")
            ];

            $payments[] = $payment;

            file_put_contents(
                $paymentFile,
                json_encode($payments, JSON_PRETTY_PRINT)
            );

            $message = "Rent paid successfully for " . $student["name"];

            break;
        }
    }
}


// ===============================
// DELETE STUDENT
// ===============================

if (isset($_GET["delete"])) {

    $deleteId = $_GET["delete"];

    $newStudents = [];

    foreach ($students as $student) {

        if ($student["id"] != $deleteId) {
            $newStudents[] = $student;
        }
    }

    $students = $newStudents;

    file_put_contents(
        $studentFile,
        json_encode($students, JSON_PRETTY_PRINT)
    );

    $message = "Student deleted successfully!";
}


// ===============================
// CHECK IF RENT PAID THIS MONTH
// ===============================

function rentPaid($studentId, $payments)
{
    $currentMonth = date("Y-m");

    foreach ($payments as $payment) {

        if (
            $payment["student_id"] == $studentId &&
            $payment["rent_month"] == $currentMonth
        ) {
            return true;
        }
    }

    return false;
}


// ===============================
// RENT NOTIFICATION
// ===============================

function rentNotification($student, $payments)
{
    if (rentPaid($student["id"], $payments)) {

        return "<span class='paid'>✓ Rent Paid</span>";
    }

    $today = date("Y-m-d");

    $dueDay = str_pad(
        $student["due_day"],
        2,
        "0",
        STR_PAD_LEFT
    );

    $dueDate = date("Y-m") . "-" . $dueDay;

    $daysLeft =
        (strtotime($dueDate) - strtotime($today)) / 86400;

    if ($daysLeft < 0) {

        return "<span class='overdue'>
                ⚠ Rent Overdue
                </span>";

    } elseif ($daysLeft == 0) {

        return "<span class='today'>
                🔔 Rent Due Today
                </span>";

    } elseif ($daysLeft <= 5) {

        return "<span class='soon'>
                🔔 Due in " .
                ceil($daysLeft) .
                " days
                </span>";

    } else {

        return "<span class='normal'>
                Rent Not Due
                </span>";
    }
}
